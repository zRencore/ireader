# Dependencias para Lector de PDF con Voz
# Todas las librerías son gratuitas y de código abierto.
# Instalar con:  pip install -r requirements.txt
#
# Las versiones se especifican con >= para permitir parches de seguridad
# pero garantizando compatibilidad con la API usada en el código.

# --- Interfaz de usuario ---
streamlit>=1.30.0,<2.0.0     # Framework principal para la UI web local

# --- Procesamiento de PDF ---
pypdf>=4.0.0,<6.0.0          # Extracción de texto de PDFs digitales (MIT)

# --- Motor de texto a voz (Edge-TTS - PREFERIDO, máxima calidad) ---
edge-tts>=6.1.0              # Voces neurales Microsoft Edge Read Aloud (español)
                            # Requiere internet. Salida MP3.

# --- Motor de texto a voz (Piper TTS - offline, calidad media) ---
# NOTA: piper-tts y onnxruntime son opcionales. La app funciona sin ellos
#       usando Edge-TTS o pyttsx3. Se instalan solo si se quiere Piper TTS.
piper-tts>=1.2.0             # Voces neuronales locales en español
onnxruntime>=1.16.0          # Backend de inferencia para modelos Piper

# --- Procesamiento de audio ---
soundfile>=0.12.0            # Lectura/escritura de archivos WAV
numpy>=1.24.0                # Manipulación de arrays de audio

# --- Motor de respaldo (Windows SAPI5) ---
pyttsx3>=2.90                # Fallback usando voces nativas de Windows

# --- Utilidades ---
requests>=2.31.0             # Descarga de voces Piper desde HuggingFace
